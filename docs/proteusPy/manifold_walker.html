proteusPy.manifold_walker

ManifoldWalker: Navigate embedding spaces along data manifolds using TurtleND.

Uses local PCA at each position to discover the tangent space of the data manifold, then steers the turtle along manifold-aligned directions. This replaces isotropic gradient descent with Riemannian-aware navigation that respects the actual geometry of the embedding space.

Part of the program proteusPy, https://github.com/suchanek/proteusPy, a Python package for the manipulation and analysis of macromolecules.

 27class ManifoldWalker:
 28    """Navigate an N-dimensional embedding space along its data manifold.
 29
 30    The walker maintains a TurtleND whose frame is continuously re-aligned
 31    to the local principal directions of the data via PCA on a neighborhood.
 32    Gradients are projected into this local frame, and components along
 33    low-variance (off-manifold) directions are suppressed.
 34
 35    Parameters
 36    ----------
 37    embeddings : np.ndarray
 38        The embedding matrix of shape (n_points, ndim).
 39    objective : callable
 40        A function f(position) -> scalar to minimize. The walker computes
 41        numerical gradients of this function.
 42    k : int
 43        Number of nearest neighbors for local PCA (default 50).
 44    variance_threshold : float
 45        Fraction of cumulative variance to retain (default 0.95).
 46        Directions beyond this threshold are considered off-manifold noise.
 47    learning_rate : float
 48        Step size for each move (default 0.01).
 49    epsilon : float
 50        Perturbation size for numerical gradient estimation (default 1e-5).
 51    """

 57        k: int = 50,

 59        learning_rate: float = 0.01,
 60        epsilon: float = 1e-5,
 61    ):
 62        self._embeddings = np.asarray(embeddings, dtype="d")
 63        self._n_points, self._ndim = self._embeddings.shape

 65        self._k = min(k, self._n_points - 1) if self._n_points > 1 else 1
 66        self._variance_threshold = variance_threshold
 67        self._lr = learning_rate
 68        self._epsilon = epsilon
 69
 70        self._turtle = TurtleND(self._ndim, name="ManifoldWalker")
 71
 72        # Diagnostics updated at each step
 73        self._eigenvalues = None
 74        self._intrinsic_dim = None
 75        self._history = []

129    def _knn(self, point: np.ndarray) -> np.ndarray:
130        """Return the k nearest neighbors to point from the embedding matrix."""
131        diffs = self._embeddings - point
132        dists = np.linalg.norm(diffs, axis=1)
133        indices = np.argpartition(dists, self._k)[: self._k]
134        return self._embeddings[indices]
135
136    def _local_pca(self, point: np.ndarray):
137        """
138        Compute local PCA around point.
139
140        Returns
141        -------
142        eigenvalues : np.ndarray
143            Eigenvalues in descending order.
144        eigenvectors : np.ndarray
145            Corresponding eigenvectors as columns, reordered descending.
146        intrinsic_dim : int
147            Number of dimensions capturing variance_threshold of variance.
148        """
149        neighbors = self._knn(point)
150        centered = neighbors - neighbors.mean(axis=0)
151        cov = (centered.T @ centered) / (len(neighbors) - 1)
152
153        eigenvalues, eigenvectors = np.linalg.eigh(cov)
154
155        # eigh returns ascending order; flip to descending
156        eigenvalues = eigenvalues[::-1]
157        eigenvectors = eigenvectors[:, ::-1]
158
159        # Clamp negative eigenvalues (numerical noise)
160        eigenvalues = np.maximum(eigenvalues, 0.0)
161
162        # Determine intrinsic dimensionality from cumulative variance
163        total = eigenvalues.sum()
164        if total > 0:
165            cumulative = np.cumsum(eigenvalues) / total
166            intrinsic_dim = int(np.searchsorted(cumulative, self._variance_threshold) + 1)
167        else:
168            intrinsic_dim = self._ndim
169
170        return eigenvalues, eigenvectors, intrinsic_dim
171
172    def _numerical_gradient(self, point: np.ndarray) -> np.ndarray:
173        """Compute the gradient of the objective via central differences."""
174        grad = np.zeros(self._ndim, dtype="d")
175        for i in range(self._ndim):
176            perturb = np.zeros(self._ndim, dtype="d")
177            perturb[i] = self._epsilon
178            grad[i] = (self._objective(point + perturb) - self._objective(point - perturb)) / (
179                2 * self._epsilon
180            )
181        return grad
182
183    def orient(self) -> int:
184        """
185        Re-orient the turtle's frame to the local principal directions.
186
187        Computes local PCA around the current position and aligns the
188        turtle's basis vectors to the eigenvectors (descending eigenvalue).
189
190        Returns
191        -------
192        intrinsic_dim : int
193            The estimated local intrinsic dimensionality.
194        """
195        eigenvalues, eigenvectors, intrinsic_dim = self._local_pca(self._turtle._position)
196        self._eigenvalues = eigenvalues
197        self._intrinsic_dim = intrinsic_dim
198
199        # Set the turtle's frame: rows are basis vectors = eigenvector columns transposed
200        self._turtle._frame = eigenvectors.T.copy()
201
202        return intrinsic_dim

204    def step(self, gradient: np.ndarray | None = None) -> float:
205        """
206        Take one manifold-aware step.
207
208        1. Compute local PCA and orient the turtle
209        2. Compute or accept the objective gradient
210        3. Project gradient into the local frame
211        4. Zero out off-manifold components
212        5. Step along the remaining manifold-aligned directions
213
214        Parameters
215        ----------
216        gradient : np.ndarray, optional
217            Pre-computed gradient in global coordinates. If None, the
218            numerical gradient of the objective is computed.
219
220        Returns
221        -------
222        objective_value : float
223            The objective value at the new position.
224        """
225        pos = self._turtle._position.copy()
226
227        # 1. Orient to local manifold
228        intrinsic_dim = self.orient()
229
230        # 2. Get gradient
231        if gradient is None:
232            raw_grad = self._numerical_gradient(pos)
233        else:
234            raw_grad = np.asarray(gradient, dtype="d")
235
236        # 3. Project gradient into local (PCA-aligned) frame
237        local_grad = self._turtle._frame @ raw_grad
238
239        # 4. Zero out off-manifold components
240        local_grad[intrinsic_dim:] = 0.0
241
242        # 5. Weight by eigenvalue magnitude (natural gradient-style)
243        # Directions with more variance get proportionally more trust
244        if self._eigenvalues is not None and self._eigenvalues[0] > 0:
245            weights = np.zeros(self._ndim, dtype="d")
246            weights[:intrinsic_dim] = self._eigenvalues[:intrinsic_dim] / self._eigenvalues[0]
247            local_grad *= weights
248
249        # 6. Convert back to global and step (negative = descent)
250        global_step = self._turtle._frame.T @ local_grad
251        self._turtle._position = pos - self._lr * global_step
252
253        obj_val = self._objective(self._turtle._position)
254        self._history.append((self._turtle._position.copy(), obj_val))
255
256        return obj_val
257
258    def walk(
259        self,
260        n_steps: int = 100,
261        gradient_fn: Callable[[np.ndarray], np.ndarray] | None = None,
262        tol: float = 1e-8,
263        verbose: bool = False,
264    ) -> np.ndarray:
265        """
266        Walk along the manifold for n_steps, minimizing the objective.
267
268        Parameters
269        ----------
270        n_steps : int
271            Maximum number of steps.
272        gradient_fn : callable, optional
273            A function position -> gradient. If provided, used instead of
274            numerical differentiation. Use this when you have analytic
275            gradients (e.g. from autograd / torch).
276        tol : float
277            Stop if the objective changes by less than tol between steps.
278        verbose : bool
279            Print progress every 10 steps.
280
281        Returns
282        -------
283        final_position : np.ndarray
284            The position after walking.
285        """
286        prev_val = self._objective(self._turtle._position)
287
288        for i in range(n_steps):
289            grad = gradient_fn(self._turtle._position) if gradient_fn else None
290            val = self.step(gradient=grad)
291
292            if verbose and i % 10 == 0:
293                print(
294                    f"Step {i:4d}: obj={val:.6e}  intrinsic_dim={self._intrinsic_dim}/{self._ndim}"
295                )
296
297            if abs(val - prev_val) < tol:
298                if verbose:
299                    print(f"Converged at step {i} (delta={abs(val - prev_val):.2e})")
300                break
301            prev_val = val
302
303        return self._turtle._position.copy()
304
305    def probe(self, direction_index: int, distances: np.ndarray) -> np.ndarray:
306        """
307        Probe the objective along a specific basis direction.
308
309        Useful for understanding the loss landscape along individual
310        principal components without moving the turtle.
311
312        Parameters
313        ----------
314        direction_index : int
315            Which basis vector to probe along (0 = highest variance).
316        distances : np.ndarray
317            Array of signed distances to evaluate at.
318
319        Returns
320        -------
321        values : np.ndarray
322            Objective values at each probed point.
323        """
324        pos = self._turtle._position.copy()
325        direction = self._turtle._frame[direction_index]
326        values = np.array(
327            [self._objective(pos + d * direction) for d in distances],
328            dtype="d",
329        )
330        return values

341class ManifoldAdamWalker(ManifoldWalker):
342    """ManifoldWalker with Adam-style momentum and adaptive LR in the projected subspace.
343
344    The key insight: Adam's machinery (momentum + per-dimension adaptive LR)
345    operates entirely within the d-dimensional manifold tangent space, not
346    in the full N-dimensional ambient space.  Off-manifold gradient components
347    are suppressed *before* the adaptive update, so momentum never accumulates
348    noise and the adaptive denominator only tracks signal variance.
349
350    This inverts the eigenvalue weighting of the base ManifoldWalker:
351    instead of trusting high-variance directions more, the Adam denominator
352    naturally takes smaller steps in steep (high-gradient-variance) directions
353    and larger steps in flat directions — which is correct for optimization.
354
355    Parameters
356    ----------
357    embeddings : np.ndarray
358        Embedding matrix (n_points, ndim).
359    objective : callable
360        f(position) -> scalar to minimize.
361    k : int
362        Neighborhood size for local PCA (default 50).
363    variance_threshold : float
364        Cumulative variance fraction for intrinsic-dim cutoff (default 0.95).
365    learning_rate : float
366        Step size (default 0.01).
367    epsilon : float
368        Perturbation for numerical gradients (default 1e-5).
369    beta1 : float
370        Exponential decay for first moment (default 0.9).
371    beta2 : float
372        Exponential decay for second moment (default 0.999).
373    adam_eps : float
374        Denominator epsilon for numerical stability (default 1e-7).
375    """
376
377    def __init__(
378        self,
379        embeddings: np.ndarray,
380        objective: Callable[[np.ndarray], float],
381        k: int = 50,
382        variance_threshold: float = 0.95,
383        learning_rate: float = 0.01,
384        epsilon: float = 1e-5,
385        beta1: float = 0.9,
386        beta2: float = 0.999,
387        adam_eps: float = 1e-7,
388    ):
389        super().__init__(
390            embeddings=embeddings,
391            objective=objective,
392            k=k,
393            variance_threshold=variance_threshold,
394            learning_rate=learning_rate,
395            epsilon=epsilon,
396        )
397        self._beta1 = beta1
398        self._beta2 = beta2
399        self._adam_eps = adam_eps
400
401        # Adam state — lives in full N-dim space but only d dims are active.
402        # Accumulated in *global* coordinates so the state persists across
403        # frame re-orientations.
404        self._m = np.zeros(self._ndim, dtype="d")  # first moment
405        self._v = np.zeros(self._ndim, dtype="d")  # second moment
406        self._t = 0
407
408    def step(self, gradient: np.ndarray | None = None) -> float:
409        """Take one manifold-aware Adam step.
410
411        1. Orient turtle to local manifold via PCA
412        2. Project gradient into local frame, zero off-manifold components
413        3. Map projected gradient back to global coordinates
414        4. Run Adam update on the manifold-projected gradient
415        5. Step
416
417        Parameters
418        ----------
419        gradient : np.ndarray, optional
420            Pre-computed gradient in global coordinates.
421
422        Returns
423        -------
424        objective_value : float
425            Objective at the new position.
426        """
427        pos = self._turtle._position.copy()
428
429        # 1. Orient to local manifold
430        intrinsic_dim = self.orient()
431
432        # 2. Get gradient
433        if gradient is None:
434            raw_grad = self._numerical_gradient(pos)
435        else:
436            raw_grad = np.asarray(gradient, dtype="d")
437
438        # 3. Project into local frame and suppress off-manifold
439        local_grad = self._turtle._frame @ raw_grad
440        local_grad[intrinsic_dim:] = 0.0
441        # No eigenvalue weighting — let Adam handle step-size adaptation
442
443        # 4. Map back to global coordinates (manifold-projected gradient)
444        projected_grad = self._turtle._frame.T @ local_grad
445
446        # 5. Adam update on the projected gradient
447        self._t += 1
448        self._m = self._beta1 * self._m + (1 - self._beta1) * projected_grad
449        self._v = self._beta2 * self._v + (1 - self._beta2) * (projected_grad**2)
450
451        m_hat = self._m / (1 - self._beta1**self._t)
452        v_hat = self._v / (1 - self._beta2**self._t)
453
454        adam_step = m_hat / (np.sqrt(v_hat) + self._adam_eps)
455
456        # 6. Descend
457        self._turtle._position = pos - self._lr * adam_step
458
459        obj_val = self._objective(self._turtle._position)
460        self._history.append((self._turtle._position.copy(), obj_val))
461
462        return obj_val
463
464    def __repr__(self):
465        id_str = f"intrinsic_dim={self._intrinsic_dim}" if self._intrinsic_dim else "not oriented"
466        return (
467            f"<ManifoldAdamWalker: {self._ndim}D, {self._n_points} embeddings, "
468            f"k={self._k}, lr={self._lr}, {id_str}, "
469            f"t={self._t}, {len(self._history)} steps taken>"
470        )
