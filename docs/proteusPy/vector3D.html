def calc_dihedral( v1: turtlend.vector3D.Vector3D, v2: turtlend.vector3D.Vector3D, v3: turtlend.vector3D.Vector3D, v4: turtlend.vector3D.Vector3D) -> float: 
254def calc_dihedral(v1: Vector3D, v2: Vector3D, v3: Vector3D, v4: Vector3D) -> float:
255    """
256    Return the dihedral angle between four 3D points, (-180-180 degrees).
257
258    :param v1: The first point.
259    :param v2: The second point.
260    :param v3: The third point.
261    :param v4: The fourth point.
262    :return: The dihedral angle in degrees.
263    :rtype: float
264
265    :example:
266    >>> v1 = Vector3D(0.0, 0.0, 0.0)
267    >>> v2 = Vector3D(1.0, 0.0, 0.0)
268    >>> v3 = Vector3D(1.0, 1.0, 1.0)
269    >>> v4 = Vector3D(1.0, 1.0, 2.0)
270    >>> float(round(calc_dihedral(v1, v2, v3, v4), 2))
271    90.0
272    """
273    ab = v1 - v2
274    cb = v3 - v2
275    db = v4 - v3
276    u = ab**cb
277    v = db**cb
278    w = u**v
279    angle = u.angle_with(v)
280    try:
281        if cb.angle_with(w) > 0.001:
282            angle = -angle
283    except ZeroDivisionError:
284        pass
285    return angle

def calculate_bond_angle(atom1, atom2, atom3): 
331def calculate_bond_angle(atom1, atom2, atom3):
332    """
333    Calculate the bond angle between three atoms given their x, y, z coordinates.
334
335    :param atom1: x, y, z coordinates of the first atom.
336    :param atom2: x, y, z coordinates of the second atom.
337    :param atom3: x, y, z coordinates of the third atom.
338    :return: Bond angle in degrees.
339    :rtype: float
340    """
341    atom1 = np.array(atom1)
342    atom2 = np.array(atom2)
343    atom3 = np.array(atom3)
344
345    vector1 = atom1 - atom2
346    vector2 = atom3 - atom2
347
348    dot_product = np.dot(vector1, vector2)
349    magnitude1 = np.linalg.norm(vector1)
350    magnitude2 = np.linalg.norm(vector2)
351
352    if magnitude1 == 0 or magnitude2 == 0:
353        _logger.warning("Cannot calculate angle with zero vector")
354        return 0.0
355
356    cos_angle = dot_product / (magnitude1 * magnitude2)
357    angle_radians = np.arccos(np.clip(cos_angle, -1.0, 1.0))
358    return np.degrees(angle_radians)

def rms_difference( calculated_angles: numpy.ndarray, idealized_angles: numpy.ndarray) -> float: 
310def rms_difference(calculated_angles: np.ndarray, idealized_angles: np.ndarray) -> float:
311    """
312    Calculate the Root Mean Square (RMS) difference between two angle arrays.
313
314    :param calculated_angles: An array of calculated angles.
315    :param idealized_angles: An array of idealized angles.
316    :return: The RMS difference between the calculated and idealized angles.
317    :rtype: float
318    :raises ValueError: If the input arrays do not have the same shape or are empty.
319    """
320    if calculated_angles.shape != idealized_angles.shape:
321        raise ValueError("Input arrays must have the same shape")
322    if calculated_angles.size == 0 or idealized_angles.size == 0:
323        raise ValueError("Input arrays must not be empty")
324
325    differences = np.subtract(calculated_angles, idealized_angles)
326    squared_differences = np.square(differences)
327    mean_squared_difference = np.mean(squared_differences)
328    return np.sqrt(mean_squared_difference)
