proteusPy.tree_visualizer

TreeVisualizer: Text, PNG, and 3D rendering of DisulfideTree hierarchies.

Provides three output modes:

  1. Text — Unicode box-drawing for terminal / print / markdown
  2. PNG — matplotlib schematic for publication / slides
  3. 3D — interactive pyvista scene with spheres, cylinders, and labels

All renderers walk the DisulfideTree via its children() / members() / node_data() API and require no additional graph libraries.

Part of the program proteusPy, https://github.com/suchanek/proteusPy, a Python package for the manipulation and analysis of macromolecules.

Author: Eric G. Suchanek, PhD

 32_PIPE = "\u2502"  # │
 33_TEE = "\u251c"  # ├
 34_ELBOW = "\u2514"  # └
 35_DASH = "\u2500"  # ─
 36_SPACE = " "

 39def _format_node_label(
 40    node_key: str,
 41    data: TreeNodeData | None,
 42    show_occupancy: bool = True,
 43    show_pct: bool = True,
 44) -> str:
 45    """Build a human-readable label for one tree node."""
 46    if node_key == "root":
 47        if data:
 48            return f"root ({data.occupancy:,} SS)"
 49        return "root"
 50
 51    parts = [node_key]
 52    if data:
 53        if data.class_name:
 54            parts.append(data.class_name)
 55        if show_occupancy:
 56            parts.append(f"n={data.occupancy:,}")
 57        if show_pct:
 58            parts.append(f"{data.occupancy_pct:.1f}%")
 59    return " ".join(parts)

 62def _text_tree_recursive(
 63    tree: DisulfideTree,
 64    node_key: str,
 65    prefix: str,
 66    is_last: bool,
 67    lines: list[str],
 68    max_children: int,
 69    max_depth: int,
 70    current_depth: int,
 71    show_members: bool,
 72    show_occupancy: bool,
 73    show_pct: bool,
 74) -> None:
 75    """Recursively build text tree lines."""
 76    data = tree.node_data(node_key)
 77    label = _format_node_label(node_key, data, show_occupancy, show_pct)
 78
 79    # Connector for this node
 80    if current_depth == 0:
 81        lines.append(label)
 82    else:
 83        connector = f"{_ELBOW}{_DASH}{_DASH} " if is_last else f"{_TEE}{_DASH}{_DASH} "
 84        lines.append(f"{prefix}{connector}{label}")
 85
 86    # Update prefix for children
 87    if current_depth == 0:
 88        child_prefix = ""
 89    else:
 90        child_prefix = prefix + (_SPACE * 4 if is_last else f"{_PIPE}   ")
 91
 92    if current_depth >= max_depth:
 93        return
 94
 95    # Get children (hierarchy edges)
 96    children = tree.children(node_key)
 97
 98    # Optionally show leaf members
 99    members = []
100    if show_members and not children:
101        members = tree.members(node_key)
102
103    all_items = children + members
104    truncated = False
105    if max_children > 0 and len(all_items) > max_children:
106        all_items = all_items[:max_children]
107        truncated = True
108
109    for i, child_key in enumerate(all_items):
110        is_last_child = (i == len(all_items) - 1) and not truncated
111        if child_key in [m for m in members]:
112            # Leaf member — just show the ID
113            connector = (
114                f"{_ELBOW}{_DASH}{_DASH} "
115                if is_last_child and not truncated
116                else f"{_TEE}{_DASH}{_DASH} "
117            )
118            lines.append(f"{child_prefix}{connector}{child_key}")
119        else:
120            _text_tree_recursive(
121                tree,
122                child_key,
123                child_prefix,
124                is_last_child,
125                lines,
126                max_children,
127                max_depth,
128                current_depth + 1,
129                show_members,
130                show_occupancy,
131                show_pct,
132            )
133
134    if truncated:
135        remaining = len(children) + len(members) - max_children
136        connector = f"{_ELBOW}{_DASH}{_DASH} "
137        lines.append(f"{child_prefix}{connector}... +{remaining} more")

140def text_tree(
141    tree: DisulfideTree,
142    root: str = "root",
143    max_children: int = 8,
144    max_depth: int = 4,
145    show_members: bool = False,
146    show_occupancy: bool = True,
147    show_pct: bool = True,
148) -> str:
149    """Render a DisulfideTree as a Unicode text tree.
150
151    Parameters
152    ----------
153    tree : DisulfideTree
154        The tree to render.
155    root : str
156        Starting node key (default ``"root"``).
157    max_children : int
158        Maximum children to show per node (0 = unlimited).
159    max_depth : int
160        Maximum depth to descend (0 = root only, 4 = full hierarchy).
161    show_members : bool
162        Whether to show individual disulfide members at leaf nodes.
163    show_occupancy : bool
164        Show member counts on each node.
165    show_pct : bool
166        Show occupancy percentages on each node.
167
168    Returns
169    -------
170    str
171        Multi-line Unicode text tree.
172
173    Examples
174    --------
175    >>> print(text_tree(tree, max_depth=2, max_children=3))
176    root (36,456 SS)
177    ├── 00200b -LHSpiral n=15,234 41.8%
178    │   ├── 00200q n=8,412 23.1%
179    │   ├── 00100q n=3,201 8.8%
180    │   └── ... +5 more
181    └── 00000b n=12,100 33.2%
182    """
183    lines: list[str] = []
184    _text_tree_recursive(
185        tree,
186        root,
187        "",
188        True,
189        lines,
190        max_children,
191        max_depth,
192        0,
193        show_members,
194        show_occupancy,
195        show_pct,
196    )
197    return "\n".join(lines)

205def _collect_layout(
206    tree: DisulfideTree,
207    node_key: str,
208    depth: int,
209    max_children: int,
210    max_depth: int,
211    show_members: bool,
212    counter: list[int],
213) -> tuple[dict, list]:
214    """Walk tree and assign (x=leaf_order, y=-depth) positions.
215
216    Returns (positions dict, edges list).
217    """
218    positions = {}
219    edges = []
220
221    if depth > max_depth:
222        positions[node_key] = (counter[0], -depth)
223        counter[0] += 1
224        return positions, edges
225
226    children = tree.children(node_key)
227    members = []
228    if show_members and not children:
229        members = tree.members(node_key)
230
231    all_items = children + members
232    if max_children > 0 and len(all_items) > max_children:
233        all_items = all_items[:max_children]
234
235    if not all_items:
236        # Leaf
237        positions[node_key] = (counter[0], -depth)
238        counter[0] += 1
239        return positions, edges
240
241    child_positions = {}
242    for child_key in all_items:
243        sub_pos, sub_edges = _collect_layout(
244            tree, child_key, depth + 1, max_children, max_depth, show_members, counter
245        )
246        child_positions.update(sub_pos)
247        positions.update(sub_pos)
248        edges.extend(sub_edges)
249        edges.append((node_key, child_key))
250
251    # Parent x = mean of children x
252    child_xs = [child_positions[ck][0] for ck in all_items if ck in child_positions]
253    if child_xs:
254        parent_x = (min(child_xs) + max(child_xs)) / 2.0
255    else:
256        parent_x = counter[0]
257        counter[0] += 1
258
259    positions[node_key] = (parent_x, -depth)
260    return positions, edges

263def _node_color(data: TreeNodeData | None, node_key: str) -> str:
264    """Pick a color based on hierarchy level."""
265    if node_key == "root":
266        return "#2c3e50"
267    if data is None:
268        return "#95a5a6"  # leaf member
269    level_colors = {
270        "binary": "#e74c3c",
271        "quadrant": "#e67e22",
272        "sextant": "#f1c40f",
273        "octant": "#2ecc71",
274    }
275    return level_colors.get(data.level, "#3498db")

278def _node_label_short(node_key: str, data: TreeNodeData | None) -> str:
279    """Short label for PNG nodes."""
280    if node_key == "root":
281        return "root"
282    if data is None:
283        # Leaf member — truncate long PDB IDs
284        parts = node_key.split("_")
285        if len(parts) >= 3:
286            return f"{parts[0]}\n{parts[1]}-{parts[2]}"
287        return node_key
288    label = node_key
289    if data.class_name:
290        label += f"\n{data.class_name}"
291    return label

294def png_tree(
295    tree: DisulfideTree,
296    filename: str = "disulfide_tree.png",
297    root: str = "root",
298    max_children: int = 6,
299    max_depth: int = 3,
300    show_members: bool = False,
301    figsize: tuple[float, float] = (16, 10),
302    dpi: int = 150,
303    title: str = "Disulfide Classification Tree",
304    node_size: float = 1200,
305    font_size: int = 7,
306) -> str:
307    """Render a DisulfideTree as a PNG schematic image.
308
309    Uses matplotlib to draw a top-down hierarchical layout with
310    color-coded nodes by level and occupancy-scaled edges.
311
312    Parameters
313    ----------
314    tree : DisulfideTree
315        The tree to render.
316    filename : str
317        Output file path (supports .png, .pdf, .svg).
318    root : str
319        Starting node key.
320    max_children : int
321        Maximum children per node.
322    max_depth : int
323        Maximum depth to draw.
324    show_members : bool
325        Whether to show leaf disulfide members.
326    figsize : tuple
327        Figure size in inches (width, height).
328    dpi : int
329        Resolution for raster output.
330    title : str
331        Figure title.
332    node_size : float
333        Base node size (matplotlib scatter units).
334    font_size : int
335        Label font size.
336
337    Returns
338    -------
339    str
340        The output filename.
341    """
342    import matplotlib.patches as mpatches
343    import matplotlib.pyplot as plt
344
345    positions, edges = _collect_layout(tree, root, 0, max_children, max_depth, show_members, [0])
346
347    fig, ax = plt.subplots(1, 1, figsize=figsize)
348    ax.set_aspect("auto")
349    ax.axis("off")
350    ax.set_title(title, fontsize=14, fontweight="bold", pad=20)
351
352    # Draw edges first (below nodes)
353    for src, tgt in edges:
354        if src in positions and tgt in positions:
355            sx, sy = positions[src]
356            tx, ty = positions[tgt]
357
358            # Get edge weight for line thickness
359            tree.node_data(src)
360            tgt_data = tree.node_data(tgt)
361            weight = 1.0
362            if tgt_data and tgt_data.occupancy_pct > 0:
363                weight = max(0.5, min(4.0, tgt_data.occupancy_pct / 5.0))
364
365            # Orthogonal routing: vertical down, then horizontal, then vertical
366            mid_y = (sy + ty) / 2.0
367            ax.plot(
368                [sx, sx, tx, tx],
369                [sy, mid_y, mid_y, ty],
370                color="#bdc3c7",
371                linewidth=weight,
372                solid_capstyle="round",
373                zorder=1,
374            )
375
376    # Draw nodes
377    for node_key, (x, y) in positions.items():
378        data = tree.node_data(node_key)
379        color = _node_color(data, node_key)
380        label = _node_label_short(node_key, data)
381
382        # Scale node size by occupancy
383        size = node_size
384        if data and data.occupancy_pct > 0:
385            size = max(600, min(3000, node_size * (data.occupancy_pct / 10.0)))
386
387        ax.scatter(x, y, s=size, c=color, zorder=3, edgecolors="white", linewidths=1.5)
388        ax.text(
389            x,
390            y,
391            label,
392            ha="center",
393            va="center",
394            fontsize=font_size,
395            color="white",
396            fontweight="bold",
397            zorder=4,
398        )
399
400        # Occupancy annotation below node
401        if data and data.occupancy > 0 and node_key != "root":
402            ax.text(
403                x,
404                y - 0.35,
405                f"n={data.occupancy:,}",
406                ha="center",
407                va="top",
408                fontsize=font_size - 1,
409                color="#7f8c8d",
410                zorder=4,
411            )
412
413    # Legend
414    legend_items = [
415        mpatches.Patch(color="#2c3e50", label="Root"),
416        mpatches.Patch(color="#e74c3c", label="Binary (2\u2075)"),
417        mpatches.Patch(color="#e67e22", label="Quadrant (4\u2075)"),
418        mpatches.Patch(color="#f1c40f", label="Sextant (6\u2075)"),
419        mpatches.Patch(color="#2ecc71", label="Octant (8\u2075)"),
420    ]
421    if show_members:
422        legend_items.append(mpatches.Patch(color="#95a5a6", label="Member"))
423
424    ax.legend(
425        handles=legend_items,
426        loc="lower right",
427        fontsize=8,
428        framealpha=0.9,
429    )
430
431    plt.tight_layout()
432    fig.savefig(filename, dpi=dpi, bbox_inches="tight", facecolor="white")
433    plt.close(fig)
434    return filename

441#: Level colors as RGB floats for pyvista meshes.
442_LEVEL_COLORS_RGB = {
443    "root": (0.173, 0.243, 0.314),  # #2c3e50
444    "binary": (0.906, 0.298, 0.235),  # #e74c3c
445    "quadrant": (0.902, 0.494, 0.133),  # #e67e22
446    "sextant": (0.945, 0.769, 0.059),  # #f1c40f
447    "octant": (0.180, 0.800, 0.443),  # #2ecc71
448    "member": (0.584, 0.647, 0.651),  # #95a5a6
449}
450
451#: Y spacing between hierarchy levels in the 3D scene.
452_LEVEL_Y_SPACING = 3.0
453
454#: Base sphere radius for nodes.
455_BASE_SPHERE_RADIUS = 0.3
456
457#: Cylinder radius for edges.
458_EDGE_RADIUS = 0.04

461def _collect_layout_3d(
462    tree: DisulfideTree,
463    node_key: str,
464    depth: int,
465    max_children: int,
466    max_depth: int,
467    show_members: bool,
468    counter: list[float],
469    x_spacing: float,
470) -> tuple[dict[str, np.ndarray], list[tuple[str, str]]]:
471    """Walk tree and assign 3D positions (x=leaf_order, y=-depth, z=0).
472
473    Returns (positions dict, edges list).
474    """
475    positions: dict[str, np.ndarray] = {}
476    edges: list[tuple[str, str]] = []
477
478    if depth > max_depth:
479        positions[node_key] = np.array([counter[0] * x_spacing, -depth * _LEVEL_Y_SPACING, 0.0])
480        counter[0] += 1
481        return positions, edges
482
483    children = tree.children(node_key)
484    members = []
485    if show_members and not children:
486        members = tree.members(node_key)
487
488    all_items = children + members
489    if max_children > 0 and len(all_items) > max_children:
490        all_items = all_items[:max_children]
491
492    if not all_items:
493        positions[node_key] = np.array([counter[0] * x_spacing, -depth * _LEVEL_Y_SPACING, 0.0])
494        counter[0] += 1
495        return positions, edges
496
497    child_positions: dict[str, np.ndarray] = {}
498    for child_key in all_items:
499        sub_pos, sub_edges = _collect_layout_3d(
500            tree,
501            child_key,
502            depth + 1,
503            max_children,
504            max_depth,
505            show_members,
506            counter,
507            x_spacing,
508        )
509        child_positions.update(sub_pos)
510        positions.update(sub_pos)
511        edges.extend(sub_edges)
512        edges.append((node_key, child_key))
513
514    # Parent x = mean of children x
515    child_xs = [child_positions[ck][0] for ck in all_items if ck in child_positions]
516    if child_xs:
517        parent_x = (min(child_xs) + max(child_xs)) / 2.0
518    else:
519        parent_x = counter[0] * x_spacing
520        counter[0] += 1
521
522    positions[node_key] = np.array([parent_x, -depth * _LEVEL_Y_SPACING, 0.0])
523    return positions, edges

526def _node_color_rgb(data: TreeNodeData | None, node_key: str) -> tuple[float, float, float]:
527    """Return an RGB tuple for a tree node."""
528    if node_key == "root":
529        return _LEVEL_COLORS_RGB["root"]
530    if data is None:
531        return _LEVEL_COLORS_RGB["member"]
532    return _LEVEL_COLORS_RGB.get(data.level, (0.204, 0.596, 0.859))

535def _node_label_3d(node_key: str, data: TreeNodeData | None) -> str:
536    """Compact label for 3D point labels."""
537    if node_key == "root":
538        if data:
539            return f"root ({data.occupancy:,})"
540        return "root"
541    if data is None:
542        return node_key
543    label = node_key
544    if data.class_name:
545        label += f" {data.class_name}"
546    label += f" n={data.occupancy:,}"
547    return label

550def tree_3d(
551    tree: DisulfideTree,
552    root: str = "root",
553    max_children: int = 6,
554    max_depth: int = 3,
555    show_members: bool = False,
556    title: str = "Disulfide Classification Tree",
557    x_spacing: float = 2.0,
558    shadows: bool = False,
559    light: str = "auto",
560    off_screen: bool = False,
561    screenshot: str | None = None,
562    window_size: tuple[int, int] = (1024, 1024),
563) -> str | None:
564    """Render a DisulfideTree as an interactive 3D pyvista scene.
565
566    Nodes are drawn as spheres sized by occupancy percentage, colored
567    by hierarchy level.  Edges are drawn as cylinders with thickness
568    proportional to child occupancy.  Labels are placed above each
569    sphere.
570
571    Parameters
572    ----------
573    tree : DisulfideTree
574        The tree to render.
575    root : str
576        Starting node key.
577    max_children : int
578        Maximum children per node.
579    max_depth : int
580        Maximum depth to draw.
581    show_members : bool
582        Whether to show leaf disulfide members.
583    title : str
584        Scene title.
585    x_spacing : float
586        Horizontal spacing between leaf nodes.
587    shadows : bool
588        Enable shadow rendering.
589    light : str
590        PyVista theme: ``"auto"``, ``"light"``, or ``"dark"``.
591    off_screen : bool
592        If True, render without displaying a window (for screenshots).
593    screenshot : str, optional
594        If given, save a screenshot to this path and close. Implies
595        ``off_screen=True``.
596    window_size : tuple[int, int]
597        Window size in pixels.
598
599    Returns
600    -------
601    str or None
602        The screenshot filename if saved, else None.
603    """
604    import pyvista as pv
605
606    from proteusPy.utility import dpi_adjusted_fontsize, set_pyvista_theme
607
608    set_pyvista_theme(light)
609
610    if screenshot:
611        off_screen = True
612
613    # Collect positions and edges
614    positions, edges = _collect_layout_3d(
615        tree, root, 0, max_children, max_depth, show_members, [0], x_spacing
616    )
617
618    pl = pv.Plotter(window_size=window_size, off_screen=off_screen)
619    pl.add_title(title=title, font_size=dpi_adjusted_fontsize(10))
620    pl.enable_anti_aliasing("msaa")
621
622    # --- Draw edges as cylinders ---
623    for src, tgt in edges:
624        if src not in positions or tgt not in positions:
625            continue
626
627        src_pos = positions[src]
628        tgt_pos = positions[tgt]
629
630        direction = tgt_pos - src_pos
631        height = float(np.linalg.norm(direction))
632        if height < 1e-6:
633            continue
634
635        midpoint = (src_pos + tgt_pos) / 2.0
636
637        # Edge thickness from child occupancy
638        tgt_data = tree.node_data(tgt)
639        edge_radius = _EDGE_RADIUS
640        if tgt_data and tgt_data.occupancy_pct > 0:
641            edge_radius = max(0.02, min(0.12, _EDGE_RADIUS * tgt_data.occupancy_pct / 3.0))
642
643        cyl = pv.Cylinder(
644            center=midpoint.tolist(),
645            direction=direction.tolist(),
646            radius=edge_radius,
647            height=height,
648            capping=True,
649            resolution=16,
650        )
651        pl.add_mesh(cyl, color="#bdc3c7", smooth_shading=True)
652
653    # --- Draw nodes as spheres ---
654    for node_key, pos in positions.items():
655        data = tree.node_data(node_key)
656        color = _node_color_rgb(data, node_key)
657        label = _node_label_3d(node_key, data)
658
659        # Scale radius by occupancy
660        radius = _BASE_SPHERE_RADIUS
661        if data and data.occupancy_pct > 0:
662            radius = max(
663                0.15,
664                min(0.8, _BASE_SPHERE_RADIUS * (data.occupancy_pct / 5.0)),
665            )
666
667        sphere = pv.Sphere(
668            center=pos.tolist(),
669            radius=radius,
670            theta_resolution=24,
671            phi_resolution=24,
672        )
673        pl.add_mesh(
674            sphere,
675            color=color,
676            smooth_shading=True,
677            specular=0.7,
678            specular_power=80,
679        )
680
681        # Label above sphere
682        label_pos = pos.copy()
683        label_pos[1] += radius + 0.2
684        pl.add_point_labels(
685            [label_pos.tolist()],
686            [label],
687            font_size=8,
688            point_size=0,
689            bold=True,
690            text_color=color,
691            shape_opacity=0.0,
692            always_visible=True,
693        )
694
695    # Camera: isometric view from above-front
696    pl.camera_position = "iso"
697    pl.reset_camera()
698
699    if shadows:
700        pl.enable_shadows()
701
702    if screenshot:
703        try:
704            pl.screenshot(screenshot)
705        except RuntimeError:
706            pass
707        pl.close()
708        return screenshot
709
710    pl.show()
711    return None

336        Label font size.

338    Returns

341        The output filename.
342    """
343    import matplotlib.patches as mpatches
344    import matplotlib.pyplot as plt
345
346    positions, edges = _collect_layout(tree, root, 0, max_children, max_depth, show_members, [0])
347
348    fig, ax = plt.subplots(1, 1, figsize=figsize)
349    ax.set_aspect("auto")
350    ax.axis("off")
351    ax.set_title(title, fontsize=14, fontweight="bold", pad=20)
352
353    # Draw edges first (below nodes)
354    for src, tgt in edges:
355        if src in positions and tgt in positions:
356            sx, sy = positions[src]
357            tx, ty = positions[tgt]
358
359            # Get edge weight for line thickness
360            tree.node_data(src)
361            tgt_data = tree.node_data(tgt)
362            weight = 1.0
363            if tgt_data and tgt_data.occupancy_pct > 0:
364                weight = max(0.5, min(4.0, tgt_data.occupancy_pct / 5.0))
365
366            # Orthogonal routing: vertical down, then horizontal, then vertical
367            mid_y = (sy + ty) / 2.0
368            ax.plot(
369                [sx, sx, tx, tx],
370                [sy, mid_y, mid_y, ty],

374                zorder=1,
375            )
376

379        data = tree.node_data(node_key)
380        color = _node_color(data, node_key)
381        label = _node_label_short(node_key, data)
382
383        # Scale node size by occupancy
384        size = node_size
385        if data and data.occupancy_pct > 0:
386            size = max(600, min(3000, node_size * (data.occupancy_pct / 10.0)))
387
388        ax.scatter(x, y, s=size, c=color, zorder=3, edgecolors="white", linewidths=1.5)

395            fontsize=font_size,
396            color="white",
397            fontweight="bold",
398            zorder=4,
399        )
400
401        # Occupancy annotation below node
402        if data and data.occupancy > 0 and node_key != "root":
403            ax.text(

405                y - 0.35,

566
567    Nodes are drawn as spheres sized by occupancy percentage, colored
568    by hierarchy level.  Edges are drawn as cylinders with thickness
569    proportional to child occupancy.  Labels are placed above each
570    sphere.
571
572    Parameters
573    ----------
574    tree : DisulfideTree
575        The tree to render.
576    root : str
577        Starting node key.
578    max_children : int
579        Maximum children per node.
580    max_depth : int
581        Maximum depth to draw.
582    show_members : bool
583        Whether to show leaf disulfide members.
584    title : str
585        Scene title.
586    x_spacing : float
587        Horizontal spacing between leaf nodes.
588    shadows : bool
589        Enable shadow rendering.
590    light : str
591        PyVista theme: ``"auto"``, ``"light"``, or ``"dark"``.
592    off_screen : bool
593        If True, render without displaying a window (for screenshots).
594    screenshot : str, optional
595        If given, save a screenshot to this path and close. Implies
596        ``off_screen=True``.
597    window_size : tuple[int, int]
598        Window size in pixels.
599
600    Returns
601    -------
602    str or None
603        The screenshot filename if saved, else None.
604    """
605    import pyvista as pv
606
607    from proteusPy.utility import dpi_adjusted_fontsize, set_pyvista_theme
608
609    set_pyvista_theme(light)
610
611    if screenshot:
612        off_screen = True
613
614    # Collect positions and edges
615    positions, edges = _collect_layout_3d(
616        tree, root, 0, max_children, max_depth, show_members, [0], x_spacing
617    )
618
619    pl = pv.Plotter(window_size=window_size, off_screen=off_screen)
620    pl.add_title(title=title, font_size=dpi_adjusted_fontsize(10))
621    pl.enable_anti_aliasing("msaa")
622
623    # --- Draw edges as cylinders ---
624    for src, tgt in edges:
625        if src not in positions or tgt not in positions:
626            continue
627
628        src_pos = positions[src]
629        tgt_pos = positions[tgt]
630
631        direction = tgt_pos - src_pos
632        height = float(np.linalg.norm(direction))
633        if height < 1e-6:
634            continue
635
636        midpoint = (src_pos + tgt_pos) / 2.0
637
638        # Edge thickness from child occupancy
639        tgt_data = tree.node_data(tgt)
640        edge_radius = _EDGE_RADIUS
641        if tgt_data and tgt_data.occupancy_pct > 0:
642            edge_radius = max(0.02, min(0.12, _EDGE_RADIUS * tgt_data.occupancy_pct / 3.0))
643
644        cyl = pv.Cylinder(
645            center=midpoint.tolist(),
646            direction=direction.tolist(),
647            radius=edge_radius,
648            height=height,
649            capping=True,
650            resolution=16,
651        )
652        pl.add_mesh(cyl, color="#bdc3c7", smooth_shading=True)
653
654    # --- Draw nodes as spheres ---
655    for node_key, pos in positions.items():
656        data = tree.node_data(node_key)
657        color = _node_color_rgb(data, node_key)
658        label = _node_label_3d(node_key, data)
659
660        # Scale radius by occupancy
661        radius = _BASE_SPHERE_RADIUS
662        if data and data.occupancy_pct > 0:
663            radius = max(
664                0.15,
665                min(0.8, _BASE_SPHERE_RADIUS * (data.occupancy_pct / 5.0)),
666            )
667
668        sphere = pv.Sphere(
669            center=pos.tolist(),
670            radius=radius,
671            theta_resolution=24,
672            phi_resolution=24,
673        )
674        pl.add_mesh(
675            sphere,
676            color=color,
677            smooth_shading=True,
678            specular=0.7,
679            specular_power=80,
680        )
681
682        # Label above sphere
683        label_pos = pos.copy()
684        label_pos[1] += radius + 0.2
685        pl.add_point_labels(
686            [label_pos.tolist()],
687            [label],
688            font_size=8,
689            point_size=0,
690            bold=True,
691            text_color=color,
692            shape_opacity=0.0,
693            always_visible=True,
694        )
695
696    # Camera: isometric view from above-front
697    pl.camera_position = "iso"
698    pl.reset_camera()
699
700    if shadows:
701        pl.enable_shadows()
702
703    if screenshot:
704        try:
705            pl.screenshot(screenshot)
706        except RuntimeError:
707            pass
708        pl.close()
709        return screenshot
710
711    pl.show()
712    return None
