 34
 35    The turtle maintains:
 36      - A position vector in R^n
 37      - An orthonormal frame of n basis vectors stored as rows of an (n x n) matrix
 38
 39    By convention:
 40      - frame[0] = heading (direction of movement)
 41      - frame[1] = left
 42      - frame[2] = up  (when n >= 3)
 43      - frame[3..n-1] = higher-dimensional basis vectors

160    def _rotate(self, angle_rad: float, i: int, j: int) -> None:
161        """
162        Apply a Givens rotation in the plane of basis[i] and basis[j].
163
164        Rotates basis[i] toward basis[j] by angle_rad radians.
165        This is the fundamental rotation primitive for N dimensions.
166
167        :param angle_rad: Rotation angle in radians
168        :param i: First basis vector index
169        :param j: Second basis vector index
170        """
171        c = math.cos(angle_rad)
172        s = math.sin(angle_rad)
173
174        bi = self._frame[i].copy()
175        bj = self._frame[j].copy()
176
177        self._frame[i] = c * bi + s * bj
178        self._frame[j] = c * bj - s * bi
179
180        # re-normalize to prevent drift
181        self._frame[i] /= np.linalg.norm(self._frame[i])
182        self._frame[j] /= np.linalg.norm(self._frame[j])
183
184    def rotate(self, angle: float, i: int, j: int) -> None:
185        """
186        Rotate in the plane of basis[i] and basis[j] by angle degrees.
187
188        This is the general N-dimensional rotation. All named rotations
189        (roll, turn, pitch, yaw) are special cases.
190
191        :param angle: Rotation angle in degrees
192        :param i: First basis vector index
193        :param j: Second basis vector index
194        """
195        if i < 0 or i >= self._ndim or j < 0 or j >= self._ndim:
196            raise IndexError(f"Basis indices ({i}, {j}) out of range for {self._ndim}D")
197        if i == j:
198            raise ValueError("Cannot rotate in a degenerate plane (i == j)")
199        self._rotate(math.radians(angle), i, j)
200

205    def roll(self, angle: float) -> None:
206        """
207        Roll: rotate in the left-up plane (basis[1] toward basis[2]).
208
209        In 3D this rotates about the heading axis.
210
211        :param angle: Roll angle in degrees
212        """
213        if self._ndim < 3:
214            raise ValueError("roll requires at least 3 dimensions")
215        # Matches Turtle3D: up rotates toward -left, left rotates toward up
216        # i.e., basis[2] toward -basis[1] = basis[1] toward basis[2] with negated angle
217        # Let's match the exact Turtle3D convention:
218        #   up_new  = cos(a)*up  - sin(a)*left
219        #   left_new = cos(a)*left + sin(a)*up
220        # That's rotating basis[1] toward basis[2] by +angle
221        self._rotate(math.radians(angle), 1, 2)
222
223    def turn(self, angle: float) -> None:
224        """
225        Turn: rotate in the heading-left plane (basis[0] toward basis[1]).
226
227        In 3D this rotates about the up axis.
228
229        :param angle: Turn angle in degrees
230        """
231        self._rotate(math.radians(angle), 0, 1)
232
233    def pitch(self, angle: float) -> None:
234        """
235        Pitch: rotate in the heading-up plane (basis[0] toward basis[2]).
236
237        In 3D this rotates about the left axis. Negative angle pitches
238        the heading downward (toward -up), matching Turtle3D convention.
239
240        :param angle: Pitch angle in degrees
241        """
242        if self._ndim < 3:
243            raise ValueError("pitch requires at least 3 dimensions")
244        # Turtle3D convention:
245        #   heading_new = cos(a)*heading - sin(a)*up
246        #   up_new      = cos(a)*up      + sin(a)*heading
247        # That's basis[0] toward -basis[2], i.e., rotating by -angle in (0,2)
248        self._rotate(math.radians(-angle), 0, 2)

264    def to_local(self, global_vec) -> np.ndarray:
265        """
266        Transform a global-space vector to turtle-local coordinates.
267
268        Projects (global_vec - position) onto each basis vector.
269
270        :param global_vec: Point in global coordinates (array-like of length ndim)
271        :return: Local coordinates as ndarray of shape (ndim,)
272        """
273        global_vec = np.asarray(global_vec, dtype="d")
274        delta = global_vec - self._position
275        # Each component is the dot product with the corresponding basis vector
276        return self._frame @ delta
277
278    def to_global(self, local_vec) -> np.ndarray:
279        """
280        Transform turtle-local coordinates to global-space coordinates.
281
282        global = position + sum_i(local[i] * basis[i])
283
284        :param local_vec: Point in local coordinates (array-like of length ndim)
285        :return: Global coordinates as ndarray of shape (ndim,)
286        """
287        local_vec = np.asarray(local_vec, dtype="d")
288        # frame.T @ local_vec gives sum of local[i] * basis[i] (columns of frame.T)
289        return self._position + self._frame.T @ local_vec

295    def orthonormalize(self) -> None:
296        """
297        Re-orthonormalize the frame using modified Gram-Schmidt.
298
299        Call this periodically if accumulated floating-point drift is a concern.
300        """
301        q, _ = np.linalg.qr(self._frame.T)
302        # Preserve orientation: ensure each new basis vector is in the same
303        # half-space as the original
304        for i in range(self._ndim):
305            if np.dot(q[:, i], self._frame[i]) < 0:
306                q[:, i] = -q[:, i]
307        self._frame = q.T
308
309    def orient(self, position, heading, left) -> None:
310        """
311        Orient the turtle at position, with heading and left directions.
312        Remaining basis vectors are constructed via Gram-Schmidt.
313
314        :param position: New position (array-like of length ndim)
315        :param heading: Heading direction (array-like of length ndim)
316        :param left: Left direction (array-like of length ndim)
317        """
318        position = np.asarray(position, dtype="d")
319        heading = np.asarray(heading, dtype="d")
320        left = np.asarray(left, dtype="d")
321
322        self._position = position.copy()
323
324        # Build heading
325        h = heading - position
326        h = h / np.linalg.norm(h)
327        self._frame[0] = h
328
329        # Build left, orthogonal to heading
330        left_vec = left - position
331        left_vec = left_vec - np.dot(left_vec, h) * h
332        left_vec = left_vec / np.linalg.norm(left_vec)
333        self._frame[1] = left_vec
334
335        # Build remaining basis vectors via Gram-Schmidt from random vectors
336        # seeded deterministically
337        for k in range(2, self._ndim):
338            # Start with the k-th standard basis vector
339            v = np.zeros(self._ndim, dtype="d")
340            v[k] = 1.0
341            # Subtract projections onto all previously determined basis vectors
342            for prev in range(k):
343                v = v - np.dot(v, self._frame[prev]) * self._frame[prev]
344            norm = np.linalg.norm(v)
345            if norm < 1e-12:
346                # Fallback: try other standard basis vectors
347                for alt in range(self._ndim):
348                    v = np.zeros(self._ndim, dtype="d")
349                    v[alt] = 1.0
350                    for prev in range(k):
351                        v = v - np.dot(v, self._frame[prev]) * self._frame[prev]
352                    norm = np.linalg.norm(v)
353                    if norm > 1e-12:
354                        break
355            self._frame[k] = v / norm

361    def expand_dim(self, label: str = "") -> int:
362        """Expand the turtle's space by one dimension.
363
364        Grows position from N to N+1 (new coordinate = 0) and extends
365        the orthonormal frame with a new basis vector orthogonal to all
366        existing ones.  Returns the index of the new axis.
367
368        This is the primitive that enables temporal grounding: embed in
369        N dims, then call ``expand_dim()`` to add a time axis that the
370        turtle can navigate.
371
372        :param label: Optional human-readable label for the new axis.
373        :return: Index of the newly added dimension (== old ndim).
374        """
375        old_n = self._ndim
376        new_n = old_n + 1
377
378        # Extend position (new coordinate starts at 0)
379        new_pos = np.zeros(new_n, dtype="d")
380        new_pos[:old_n] = self._position
381
382        # Extend frame: pad each existing row with 0, add new row e_{new_n-1}
383        new_frame = np.zeros((new_n, new_n), dtype="d")
384        new_frame[:old_n, :old_n] = self._frame
385        new_frame[old_n, old_n] = 1.0  # new basis vector = standard basis
386
387        self._ndim = new_n
388        self._position = new_pos
389        self._frame = new_frame
390
391        return old_n

397    def orient_toward(self, direction) -> float:
398        """Rotate heading to face an arbitrary direction in ambient space.
399
400        Decomposes *direction* into components parallel and perpendicular
401        to the current heading, then applies a single Givens rotation in
402        the plane they span.  The rest of the frame is carried along
403        rigidly (only two basis vectors are touched).
404
405        If direction is (anti-)parallel to heading, no rotation is applied.
406
407        :param direction: Target direction vector (array-like, length ndim).
408        :return: Angle rotated in degrees.
409        """
410        d = np.asarray(direction, dtype="d")
411        norm = np.linalg.norm(d)
412        if norm < 1e-12:
413            return 0.0
414        d = d / norm
415
416        heading = self._frame[0]
417        cos_a = float(np.clip(np.dot(heading, d), -1.0, 1.0))
418
419        # Already aligned (or anti-aligned) — nothing to do
420        if abs(cos_a) > 1.0 - 1e-10:
421            return 0.0
422
423        angle_rad = math.acos(cos_a)
424
425        # Perpendicular component of d relative to heading
426        perp = d - cos_a * heading
427        perp = perp / np.linalg.norm(perp)
428
429        # Find which existing basis vector is closest to perp so we can
430        # rotate in the heading–that-basis plane.  This keeps the Givens
431        # rotation within the frame rather than introducing an arbitrary
432        # direction.
433        best_j = 1
434        best_proj = -np.inf
435        for j in range(1, self._ndim):
436            proj = abs(float(np.dot(self._frame[j], perp)))
437            if proj > best_proj:
438                best_proj = proj
439                best_j = j
440
441        # If perp doesn't align well with any single basis vector, we
442        # do a direct 2-vector rotation in the heading–perp plane.
443        # Temporarily replace basis[best_j] with perp, rotate, then
444        # re-orthonormalize to clean up.
445        self._frame[best_j] = perp
446        self._rotate(angle_rad, 0, best_j)
447        # The rotation moved heading toward perp.  Now re-orthonormalize
448        # to fix any drift from the temporary basis swap.
449        self.orthonormalize()
450
451        return math.degrees(angle_rad)

453    def orient_in_time(self, time_axis: int | None = None) -> float:
454        """Convenience: orient heading toward the temporal (last) axis.
455
456        Points toward the *negative* z-score direction of the temporal axis,
457        which corresponds to chronologically forward motion in the KNN graph.
458        The positive z-score direction is the earliest (lowest fractional-year)
459        end of the corpus; forward-in-time means decreasing z-score.
460
461        Equivalent to ``orient_toward(-e_N)`` where ``e_N`` is the unit
462        vector along axis *time_axis* (default: last dimension, as set
463        by ``expand_dim``).
464
465        :param time_axis: Axis index for time (default: ndim - 1).
466        :return: Angle rotated in degrees.
467        """
468        if time_axis is None:
469            time_axis = self._ndim - 1
470        e_t = np.zeros(self._ndim, dtype="d")
471        e_t[time_axis] = -1.0
472        return self.orient_toward(e_t)

The turtle maintains:

  • A position vector in R^n
  • An orthonormal frame of n basis vectors stored as rows of an (n x n) matrix

By convention:

  • frame[0] = heading (direction of movement)
  • frame[1] = left
  • frame[2] = up (when n >= 3)
  • frame[3..n-1] = higher-dimensional basis vectors

Rotations are Givens rotations in the plane spanned by two basis vectors. The classic 3D operations (roll, yaw, turn, pitch) are special cases.

def orient(self, position, heading, left) -> None: 
309    def orient(self, position, heading, left) -> None:
310        """
311        Orient the turtle at position, with heading and left directions.
312        Remaining basis vectors are constructed via Gram-Schmidt.
313
314        :param position: New position (array-like of length ndim)
315        :param heading: Heading direction (array-like of length ndim)
316        :param left: Left direction (array-like of length ndim)
317        """
318        position = np.asarray(position, dtype="d")
319        heading = np.asarray(heading, dtype="d")
320        left = np.asarray(left, dtype="d")
321
322        self._position = position.copy()
323
324        # Build heading
325        h = heading - position
326        h = h / np.linalg.norm(h)
327        self._frame[0] = h
328
329        # Build left, orthogonal to heading
330        left_vec = left - position
331        left_vec = left_vec - np.dot(left_vec, h) * h
332        left_vec = left_vec / np.linalg.norm(left_vec)
333        self._frame[1] = left_vec
334
335        # Build remaining basis vectors via Gram-Schmidt from random vectors
336        # seeded deterministically
337        for k in range(2, self._ndim):
338            # Start with the k-th standard basis vector
339            v = np.zeros(self._ndim, dtype="d")
340            v[k] = 1.0
341            # Subtract projections onto all previously determined basis vectors
342            for prev in range(k):
343                v = v - np.dot(v, self._frame[prev]) * self._frame[prev]
344            norm = np.linalg.norm(v)
345            if norm < 1e-12:
346                # Fallback: try other standard basis vectors
347                for alt in range(self._ndim):
348                    v = np.zeros(self._ndim, dtype="d")
349                    v[alt] = 1.0
350                    for prev in range(k):
351                        v = v - np.dot(v, self._frame[prev]) * self._frame[prev]
352                    norm = np.linalg.norm(v)
353                    if norm > 1e-12:
354                        break
355            self._frame[k] = v / norm

363
364        Grows position from N to N+1 (new coordinate = 0) and extends
365        the orthonormal frame with a new basis vector orthogonal to all
366        existing ones.  Returns the index of the new axis.
367
368        This is the primitive that enables temporal grounding: embed in
369        N dims, then call ``expand_dim()`` to add a time axis that the
370        turtle can navigate.
371
372        :param label: Optional human-readable label for the new axis.
373        :return: Index of the newly added dimension (== old ndim).
374        """
375        old_n = self._ndim
376        new_n = old_n + 1
377
378        # Extend position (new coordinate starts at 0)
379        new_pos = np.zeros(new_n, dtype="d")
380        new_pos[:old_n] = self._position
381
382        # Extend frame: pad each existing row with 0, add new row e_{new_n-1}
383        new_frame = np.zeros((new_n, new_n), dtype="d")
384        new_frame[:old_n, :old_n] = self._frame
385        new_frame[old_n, old_n] = 1.0  # new basis vector = standard basis
386
387        self._ndim = new_n
388        self._position = new_pos
389        self._frame = new_frame
390
391        return old_n

def orient_toward(self, direction) -> float: 
397    def orient_toward(self, direction) -> float:
398        """Rotate heading to face an arbitrary direction in ambient space.
399
400        Decomposes *direction* into components parallel and perpendicular
401        to the current heading, then applies a single Givens rotation in
402        the plane they span.  The rest of the frame is carried along
403        rigidly (only two basis vectors are touched).
404
405        If direction is (anti-)parallel to heading, no rotation is applied.
406
407        :param direction: Target direction vector (array-like, length ndim).
408        :return: Angle rotated in degrees.
409        """
410        d = np.asarray(direction, dtype="d")
411        norm = np.linalg.norm(d)
412        if norm < 1e-12:
413            return 0.0
414        d = d / norm
415
416        heading = self._frame[0]
417        cos_a = float(np.clip(np.dot(heading, d), -1.0, 1.0))
418
419        # Already aligned (or anti-aligned) — nothing to do
420        if abs(cos_a) > 1.0 - 1e-10:
421            return 0.0
422
423        angle_rad = math.acos(cos_a)
424
425        # Perpendicular component of d relative to heading
426        perp = d - cos_a * heading
427        perp = perp / np.linalg.norm(perp)
428
429        # Find which existing basis vector is closest to perp so we can
430        # rotate in the heading–that-basis plane.  This keeps the Givens
431        # rotation within the frame rather than introducing an arbitrary
432        # direction.
433        best_j = 1
434        best_proj = -np.inf
435        for j in range(1, self._ndim):
436            proj = abs(float(np.dot(self._frame[j], perp)))
437            if proj > best_proj:
438                best_proj = proj
439                best_j = j
440
441        # If perp doesn't align well with any single basis vector, we
442        # do a direct 2-vector rotation in the heading–perp plane.
443        # Temporarily replace basis[best_j] with perp, rotate, then
444        # re-orthonormalize to clean up.
445        self._frame[best_j] = perp
446        self._rotate(angle_rad, 0, best_j)
447        # The rotation moved heading toward perp.  Now re-orthonormalize
448        # to fix any drift from the temporary basis swap.
449        self.orthonormalize()
450
451        return math.degrees(angle_rad)
