proteusPy.disulfide_tree

DisulfideTree: Hierarchical classification tree for KGRAG integration.

Builds a multi-level tree of disulfide structural classes: binary (2^5=32) → quadrant (4^5=1024) → sextant (6^5=7776) → octant (8^5=32768) → members

Each level refines the torsion-angle classification. Branch lengths are proportional to occupancy (member count). The tree is represented as a ~proteusPy.graph_reasoner.KnowledgeGraph so the ~proteusPy.graph_reasoner.GraphReasoner can traverse it.

KGRAG integration: each node produces a structured snippet suitable for retrieval-augmented generation over protein structural knowledge.

Part of the program proteusPy, https://github.com/suchanek/proteusPy, a Python package for the manipulation and analysis of macromolecules.

 56def _segment_center(digit: int, base: int) -> float:
 57    """Return the center angle (degrees) of a segment digit for a given base.
 58
 59    Segments are numbered ``base`` down to ``1``, where digit ``base``
 60    corresponds to ``[0, 360/base)`` and digit ``1`` to
 61    ``[360 - 360/base, 360)``.
 62    """
 63    segment_size = 360.0 / base
 64    # digit = base - floor(angle / segment_size)  ⇒  angle_start = (base - digit) * segment_size
 65    angle_start = (base - digit) * segment_size
 66    return angle_start + segment_size / 2.0
 67
 68
 69def _classify_angle(angle_deg: float, base: int) -> int:
 70    """Return the segment digit for *angle_deg* at the given base.
 71
 72    For base 2, returns 0 (negative, [180, 360)) or 2 (positive, [0, 180))
 73    to match proteusPy's ``get_binary_quadrant`` convention.
 74    For other bases, returns digits 1..base via the standard formula.
 75    """
 76    angle_deg = angle_deg % 360.0
 77    if base == 2:
 78        return 2 if angle_deg < 180.0 else 0
 79    segment_size = 360.0 / base
 80    return base - int(angle_deg // segment_size)
 81
 82
 83def _digit_parent_map(child_base: int, parent_base: int) -> dict[int, int]:
 84    """Build a mapping from child digits to parent digits via center angles.
 85
 86    For each child segment, compute its center angle and classify it at
 87    the parent base.
 88    """
 89    mapping = {}
 90    for digit in range(1, child_base + 1):
 91        center = _segment_center(digit, child_base)
 92        mapping[digit] = _classify_angle(center, parent_base)
 93    return mapping
 94
 95
 96# Pre-compute the three parent maps for the hierarchy.
 97_QUADRANT_TO_BINARY = _digit_parent_map(4, 2)
 98_SEXTANT_TO_QUADRANT = _digit_parent_map(6, 4)
 99_OCTANT_TO_SEXTANT = _digit_parent_map(8, 6)
100
101_PARENT_MAPS = {
102    ("quadrant", "binary"): _QUADRANT_TO_BINARY,
103    ("sextant", "quadrant"): _SEXTANT_TO_QUADRANT,
104    ("octant", "sextant"): _OCTANT_TO_SEXTANT,
105}

108def parent_class_id(child_id: str, child_level: str, parent_level: str) -> str:
109    """Map a child class ID string to its parent class ID string.
110
111    Parameters
112    ----------
113    child_id : str
114        5-character class ID (digits only, no suffix).
115    child_level : str
116        One of ``"quadrant"``, ``"sextant"``, ``"octant"``.
117    parent_level : str
118        The next coarser level.
119
120    Returns
121    -------
122    str
123        5-character parent class ID.
124    """
125    dmap = _PARENT_MAPS[(child_level, parent_level)]
126    LEVEL_BASES[child_level]
127
128    parent_digits = []
129    for ch in child_id[:N_CHI]:
130        # Handle character encoding (base > 9 uses hex-like chars)
131        d = int(ch, 16) if ch.isalpha() else int(ch)
132        parent_digits.append(str(dmap[d]))
133    return "".join(parent_digits)

137    """Classify a set of 5 dihedral angles at the given base.
138
139    Parameters
140    ----------
141    angles : array-like
142        Five dihedral angles in degrees.
143    base : int
144        Classification base (2, 4, 6, or 8).
145
146    Returns
147    -------
148    str
149        5-character class ID.
150    """

197def snippet_for_node(node_data: TreeNodeData) -> dict:
198    """Generate a KGRAG-compatible snippet for a tree node.
199
200    Parameters
201    ----------
202    node_data : TreeNodeData
203        The tree node payload.
204
205    Returns
206    -------
207    dict
208        Structured snippet with entity_type, identifiers, properties,
209        and a natural-language description.
210    """
211    level = node_data.level
212    cid = node_data.class_id
213    suffix = LEVEL_SUFFIXES.get(level, "")
214
215    desc_parts = [f"{level.capitalize()} class {cid}{suffix}"]
216    if node_data.class_name:
217        desc_parts[0] += f" ({node_data.class_name})"
218    desc_parts.append(f"{node_data.occupancy} members ({node_data.occupancy_pct:.2f}%)")
219    if node_data.functional_annotation:
220        desc_parts.append(f"Function: {node_data.functional_annotation}")
221    if node_data.consensus_torsions:
222        angles_str = ", ".join(f"{a:.1f}" for a in node_data.consensus_torsions)
223        desc_parts.append(f"Consensus torsions: [{angles_str}]")
224
225    return {
226        "entity_type": f"disulfide_{level}_class",
227        "class_id": f"{cid}{suffix}",
228        "level": level,
229        "occupancy": node_data.occupancy,
230        "occupancy_pct": node_data.occupancy_pct,
231        "class_name": node_data.class_name or None,
232        "functional_annotation": node_data.functional_annotation or None,
233        "consensus_torsions": node_data.consensus_torsions,
234        "description": ". ".join(desc_parts) + ".",
235    }

238def snippet_for_disulfide(ss) -> dict:
239    """Generate a KGRAG-compatible snippet for an individual disulfide.
240
241    Parameters
242    ----------
243    ss : Disulfide
244        A proteusPy Disulfide object.
245
246    Returns
247    -------
248    dict
249        Structured snippet with identifiers, torsion angles, energy,
250        class memberships, and a natural-language description.
251    """
252    torsions = list(ss.torsion_array)
253    angles_str = ", ".join(f"{a:.1f}" for a in torsions)
254
255    classes = {}
256    for level, base in LEVEL_BASES.items():
257        classes[level] = classify_angles(torsions, base)
258
259    return {
260        "entity_type": "disulfide",
261        "node_id": f"{ss.pdb_id}_{ss.proximal}_{ss.distal}",
262        "pdb_id": ss.pdb_id,
263        "proximal": ss.proximal,
264        "distal": ss.distal,
265        "torsions": torsions,
266        "energy": float(ss.torsion_energy),
267        "ca_distance": float(ss.ca_distance),
268        "classes": classes,
269        "description": (
270            f"Disulfide bond in {ss.pdb_id} between "
271            f"Cys{ss.proximal}-Cys{ss.distal}: "
272            f"torsion angles [{angles_str}] deg, "
273            f"energy {ss.torsion_energy:.2f} kcal/mol, "
274            f"binary class {classes['binary']}."
275        ),
276    }

284class DisulfideTree:
285    """Hierarchical classification tree for disulfide bonds.
286
287    Builds a four-level tree (binary → quadrant → sextant → octant) with
288    individual disulfides as leaves.  Branch weights reflect occupancy.
289    The tree is stored as a :class:`KnowledgeGraph` for integration with
290    the :class:`GraphReasoner` and KGRAG.
291
292    Parameters
293    ----------
294    sslist : DisulfideList
295        Source disulfide bonds.
296    class_names : dict, optional
297        Mapping from binary class ID to human-readable name
298        (e.g. ``{"00000": "-LHSpiral"}``).
299    functional_annotations : dict, optional
300        Mapping from binary class ID to function
301        (e.g. ``{"00200": "Allosteric"}``).
302
303    Examples
304    --------
305    >>> tree = DisulfideTree(my_sslist)
306    >>> graph = tree.graph
307    >>> snippet = tree.snippet("00000b")
308    """
309
310    def __init__(
311        self,
312        sslist=None,
313        class_names: dict[str, str] | None = None,
314        functional_annotations: dict[str, str] | None = None,
315    ):
316        self._class_names = class_names or {}
317        self._functional_annotations = functional_annotations or {}
318        self._node_data: dict[str, TreeNodeData] = {}
319        self._total = 0
320
321        # The graph uses 5D embeddings (consensus torsion angles or class centers)
322        self._graph = KnowledgeGraph(ndim=N_CHI, name="disulfide_tree")
323
324        if sslist is not None:
325            self.build(sslist)
326
327    @property
328    def graph(self) -> KnowledgeGraph:
329        """The underlying knowledge graph."""
330        return self._graph
331
332    def build(self, sslist) -> None:
333        """Build the full tree from a DisulfideList.
334
335        Parameters
336        ----------
337        sslist : DisulfideList
338            Source disulfides.
339        """
340        self._total = len(sslist)
341        if self._total == 0:
342            return
343
344        # Step 1: classify every disulfide at all 4 levels
345        members: dict[str, dict[str, list]] = {level: {} for level in LEVELS}
346        member_angles: dict[str, list[np.ndarray]] = {level: {} for level in LEVELS}
347
348        for ss in sslist:
349            torsions = np.array(ss.torsion_array, dtype="d")
350            for level, base in LEVEL_BASES.items():
351                cid = classify_angles(torsions, base)
352                members[level].setdefault(cid, []).append(ss)
353                member_angles[level].setdefault(cid, []).append(torsions)
354
355        # Step 2: create nodes at each level
356        for level in LEVELS:
357            suffix = LEVEL_SUFFIXES[level]
358            for cid, ss_list in members[level].items():
359                node_key = f"{cid}{suffix}"
360                occupancy = len(ss_list)
361                occupancy_pct = 100.0 * occupancy / self._total
362
363                # Compute centroid embedding (mean of member torsion angles)
364                angle_stack = np.array(member_angles[level][cid])
365                # Circular mean for periodic angles
366                centroid = _circular_mean_deg(angle_stack)
367
368                data = TreeNodeData(
369                    class_id=cid,
370                    level=level,
371                    occupancy=occupancy,
372                    occupancy_pct=occupancy_pct,
373                    class_name=self._class_names.get(cid, ""),
374                    functional_annotation=self._functional_annotations.get(cid, ""),
375                    consensus_torsions=centroid.tolist(),
376                )
377                self._node_data[node_key] = data
378                self._graph.add_node(node_key, centroid, node=data)
379
380        # Step 3: add root node
381        all_angles = np.array([np.array(ss.torsion_array, dtype="d") for ss in sslist])
382        root_centroid = _circular_mean_deg(all_angles)
383        root_data = TreeNodeData(
384            class_id="root",
385            level="root",
386            occupancy=self._total,
387            occupancy_pct=100.0,
388        )
389        self._node_data["root"] = root_data
390        self._graph.add_node("root", root_centroid, node=root_data)
391
392        # Step 4: add hierarchical edges
393        # root → binary
394        for cid in members["binary"]:
395            node_key = f"{cid}b"
396            weight = self._node_data[node_key].occupancy_pct / 100.0
397            self._graph.add_edge(SemanticEdge("root", node_key, weight, "hierarchy"))
398
399        # binary → quadrant → sextant → octant
400        level_pairs = [
401            ("quadrant", "binary"),
402            ("sextant", "quadrant"),
403            ("octant", "sextant"),
404        ]
405        for child_level, parent_level in level_pairs:
406            child_suffix = LEVEL_SUFFIXES[child_level]
407            parent_suffix = LEVEL_SUFFIXES[parent_level]
408            for cid in members[child_level]:
409                child_key = f"{cid}{child_suffix}"
410                pid = parent_class_id(cid, child_level, parent_level)
411                parent_key = f"{pid}{parent_suffix}"
412                if parent_key in self._graph:
413                    weight = self._node_data[child_key].occupancy_pct / 100.0
414                    self._graph.add_edge(SemanticEdge(parent_key, child_key, weight, "hierarchy"))
415
416        # Step 5: add leaf edges (octant → individual disulfides)
417        for cid, ss_list in members["octant"].items():
418            oct_key = f"{cid}o"
419            for ss in ss_list:
420                member_key = f"{ss.pdb_id}_{ss.proximal}_{ss.distal}"
421                if member_key not in self._graph:
422                    emb = np.array(ss.torsion_array, dtype="d")
423                    self._graph.add_node(member_key, emb, node=ss)
424                self._graph.add_edge(SemanticEdge(oct_key, member_key, 1.0, "membership"))
425
426    def snippet(self, node_key: str) -> dict:
427        """Return a KGRAG snippet for the given node.
428
429        Parameters
430        ----------
431        node_key : str
432            A node key like ``"00000b"`` (class) or ``"1egs_24_84"`` (member).
433
434        Returns
435        -------
436        dict
437            Structured snippet for KGRAG consumption.
438        """
439        if node_key in self._node_data:
440            return snippet_for_node(self._node_data[node_key])
441
442        # Must be a leaf disulfide
443        node = self._graph.get_node(node_key)
444        if node is not None and hasattr(node, "torsion_array"):
445            return snippet_for_disulfide(node)
446
447        return {"entity_type": "unknown", "node_id": node_key, "description": ""}
448
449    def children(self, node_key: str) -> list[str]:
450        """Return the child node keys for a given node.
451
452        Parameters
453        ----------
454        node_key : str
455            Parent node key.
456
457        Returns
458        -------
459        list[str]
460            Child node keys, sorted by occupancy (descending).
461        """
462        edges = self._graph._edges.get(node_key, [])
463        child_keys = [e.target_id for e in edges if e.edge_type == "hierarchy"]
464
465        # Sort by occupancy (descending)
466        def _occ(k):
467            data = self._node_data.get(k)
468            return data.occupancy if data else 0
469
470        child_keys.sort(key=_occ, reverse=True)
471        return child_keys
472
473    def members(self, node_key: str) -> list[str]:
474        """Return leaf member keys for a given node.
475
476        Parameters
477        ----------
478        node_key : str
479            Class node key (e.g. ``"00000o"``).
480
481        Returns
482        -------
483        list[str]
484            Member disulfide node keys.
485        """
486        edges = self._graph._edges.get(node_key, [])
487        return [e.target_id for e in edges if e.edge_type == "membership"]
488
489    def node_data(self, node_key: str) -> TreeNodeData | None:
490        """Return the :class:`TreeNodeData` for *node_key*."""
491        return self._node_data.get(node_key)
492
493    @property
494    def levels(self) -> dict[str, int]:
495        """Number of populated classes at each level."""
496        counts = {level: 0 for level in LEVELS}
497        for data in self._node_data.values():
498            if data.level in counts:
499                counts[data.level] += 1
500        return counts
501
502    def __repr__(self) -> str:
503        lvl = self.levels
504        return (
505            f"DisulfideTree(total={self._total}, "
506            f"binary={lvl['binary']}, quadrant={lvl['quadrant']}, "
507            f"sextant={lvl['sextant']}, octant={lvl['octant']})"
508        )

516def _circular_mean_deg(angles: np.ndarray) -> np.ndarray:
517    """Compute the circular mean of angle arrays (in degrees).
518
519    Parameters
520    ----------
521    angles : np.ndarray
522        Shape ``(n_samples, n_angles)`` or ``(n_angles,)``.
523
524    Returns
525    -------
526    np.ndarray
527        Circular mean angle(s) in degrees, shape ``(n_angles,)``.
528    """
529    if angles.ndim == 1:
530        return angles.copy()
531    rad = np.deg2rad(angles)
532    mean_sin = np.mean(np.sin(rad), axis=0)
533    mean_cos = np.mean(np.cos(rad), axis=0)
534    return np.rad2deg(np.arctan2(mean_sin, mean_cos))
